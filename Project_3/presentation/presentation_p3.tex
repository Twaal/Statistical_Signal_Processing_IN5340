\section{Introduction}

\begin{frame}{Problem statement}
\begin{itemize}
\item Binary detection with additive Gaussian noise:
\[
H_0: x=n, \qquad H_1: x=\theta+n, \quad n\sim\mathcal{N}(0,\sigma^2)
\]
\item Use $\theta=5$ and $\sigma=5$ as baseline case.
\item Design and evaluate:
\begin{itemize}
	\item CFAR detector for given $P_{FA}$
	\item ROC curve (theoretical vs empirical)
	\item Improvement using sample mean of $N$ measurements
\end{itemize}
\end{itemize}
\end{frame}

\section{Main results}

\begin{frame}{b) Noise and signal+noise PDFs}
\begin{columns}[T,onlytextwidth]
\column{0.44\textwidth}
\small
\begin{itemize}
\item Under $H_0$: $x\sim\mathcal{N}(0,\sigma^2)$
\item Under $H_1$: $x\sim\mathcal{N}(\theta,\sigma^2)$
\item For $\theta/\sigma=1$, distributions overlap strongly.
\item Detection is possible, but not highly reliable in this regime.

\begin{frame}{e) Multi-sample detector and minimum $N$}
\small
Using sample mean of $N$ measurements:
\[
\bar{x}|H_0\sim\mathcal{N}\!\left(0,\frac{\sigma^2}{N}\right),
\qquad
\bar{x}|H_1\sim\mathcal{N}\!\left(\theta,\frac{\sigma^2}{N}\right)
\]
For CFAR with requirements $P_{FA}=0.01$ and $P_D>0.9$:
\[
N \ge \left(\frac{\sigma}{\theta}\right)^2\!\left(Q^{-1}(P_{FA}) + Q^{-1}(1-P_D)\right)^2
\]
Numerically:
\[
N_{\min,\,\text{cont}}\approx 13.02 \quad\Rightarrow\quad N_{\min}=14
\]
At $N=14$: $x_c\approx 3.11$ and $P_D\approx 0.922$.
\end{frame}

\end{itemize}
\[

\]

\begin{frame}{MAP derivation from likelihoods}
\small
For Gaussian shift model ($y|H_0\sim\mathcal N(0,\sigma^2)$, $y|H_1\sim\mathcal N(\theta,\sigma^2)$):
\[
f_0(y)=\frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{y^2}{2\sigma^2}},
\qquad
f_1(y)=\frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{(y-\theta)^2}{2\sigma^2}}
\]
\[
\log\frac{f_1(y)}{f_0(y)}
=-\frac{(y-\theta)^2}{2\sigma^2}+\frac{y^2}{2\sigma^2}
=\frac{\theta}{\sigma^2}y-\frac{\theta^2}{2\sigma^2}
\]
\vspace{0.3em}
MAP rule (equal costs):
\[
\Pr(H_1\mid y)\underset{H_0}{\overset{H_1}{\gtrless}}\Pr(H_0\mid y)
\]
\[
\frac{f_1(y)}{f_0(y)}\underset{H_0}{\overset{H_1}{\gtrless}}\frac{\Pr(H_0)}{\Pr(H_1)}=\frac{P_0}{P_1}
\]
\[
\Rightarrow\quad
y\underset{H_0}{\overset{H_1}{\gtrless}}
y_t^{MAP}=\frac{\theta}{2}+\frac{\sigma^2}{\theta}\log\frac{P_0}{P_1}
\]
\end{frame}

\begin{frame}{How priors change the detector}
\small
\begin{itemize}
\item Priors enter only through the term $\log(P_0/P_1)$ in the MAP threshold.
\item If $H_1$ is rare ($P_1<P_0$), then $\log(P_0/P_1)>0$ and threshold increases.
\item Higher threshold means fewer false alarms, but also fewer detections.
\end{itemize}
\vspace{0.4em}
\textbf{ML is a special case of MAP:}
\[
P_0=P_1\Rightarrow\log\frac{P_0}{P_1}=0
\Rightarrow y_t^{MAP}=\frac{\theta}{2}=y_t^{ML}
\]
So ML corresponds to assuming equal priors; MAP generalizes this when prior probabilities are known.
\end{frame}

\section*{Summary}

\begin{frame}{Summary}
\begin{itemize}
\item For single measurement ($\theta=\sigma=5$), overlap is substantial and $P_D$ is modest at practical $P_{FA}$.
\item CFAR with $P_{FA}=0.1$ gives threshold $x_c\approx 6.41$ and $P_D\approx 0.389$.
\item ROC simulations agree with theory in all experiments (typical AUC match within a few hundredths).
\item Averaging measurements strongly improves performance (AUC from $0.745$ at $M=1$ to $0.998$ at $M=14$).
\item With imbalanced priors, MAP shifts operating point to reduce false alarms ($0.159\rightarrow0.066$) at the cost of lower $P_D$.
\end{itemize}
\end{frame}
